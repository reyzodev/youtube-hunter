# UTF-8
#
# Version resource stamped onto YouTubeHunter.exe.
#
# Why this matters for a customer build: Windows Explorer shows these values in
# the file's Properties -> Details tab, SmartScreen and antivirus heuristics
# weigh them, and a published product whose .exe reports no publisher/version is
# both less trustworthy-looking and harder to support ("which build are you on?").
#
# Keep filevers/prodvers and the two version strings in step with
# app/constants.py (APP_VERSION) and installer.iss (MyAppVersion).
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(1, 0, 1, 0),
    prodvers=(1, 0, 1, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo(
      [
        StringTable(
          '040904B0',
          [
            StringStruct('CompanyName', 'YouTube Hunter'),
            StringStruct('FileDescription', 'YouTube Hunter - YouTube research and clip cutter'),
            StringStruct('FileVersion', '1.0.1.0'),
            StringStruct('InternalName', 'YouTubeHunter'),
            StringStruct('LegalCopyright', 'YouTube Hunter'),
            StringStruct('OriginalFilename', 'YouTubeHunter.exe'),
            StringStruct('ProductName', 'YouTube Hunter'),
            StringStruct('ProductVersion', '1.0.1.0')
          ]
        )
      ]
    ),
    VarFileInfo([VarStruct('Translation', [1033, 1200])])
  ]
)
